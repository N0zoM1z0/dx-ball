cmake_minimum_required(VERSION 3.20)
project(dxball_reconstruction VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(dxball_core SHARED src/boards.c src/resources.c src/gameplay.c src/effects.c src/particles.c src/bonuses.c src/geometry.c src/paddle.c src/round.c src/trig.c src/balls.c src/core.c src/runtime.c src/display.c src/device.c src/platform.c src/startup.c src/ui.c src/intro.c src/gameover.c src/editor.c src/midi.c src/sound.c)
if(UNIX)
    target_link_libraries(dxball_core PRIVATE m)
endif()
target_include_directories(dxball_core PUBLIC src)
set_target_properties(dxball_core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dxball_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(dxball_resources src/resource_inspector.c)
target_link_libraries(dxball_resources PRIVATE dxball_core)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dxball_resources PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(dxball_boards src/board_inspector.c)
target_link_libraries(dxball_boards PRIVATE dxball_core)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dxball_boards PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(WIN32)
    add_executable(dxball WIN32 src/windows_entry.c src/windows_adapter.c)
    set(DXBALL_RESOURCE_OBJECT "" CACHE FILEPATH "Private verified game resource object")
    if(DXBALL_RESOURCE_OBJECT)
        if(NOT EXISTS "${DXBALL_RESOURCE_OBJECT}")
            message(FATAL_ERROR "DXBALL_RESOURCE_OBJECT does not exist")
        endif()
        set_source_files_properties("${DXBALL_RESOURCE_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE)
        target_sources(dxball PRIVATE "${DXBALL_RESOURCE_OBJECT}")
    endif()
    add_executable(dxball_state_reader tests/windows_state_reader.c)
    add_executable(dxball_ddraw_loss_probe tests/windows_ddraw_loss_probe.c)
    add_executable(dxball_resource_reader tests/windows_resource_reader.c)
    set_target_properties(dxball_state_reader PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    set_target_properties(dxball_ddraw_loss_probe PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    set_target_properties(dxball_resource_reader PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_link_libraries(dxball_state_reader PRIVATE user32)
    target_link_libraries(dxball_ddraw_loss_probe PRIVATE ddraw user32)
    target_link_libraries(dxball_resource_reader PRIVATE user32 gdi32)
    target_link_libraries(dxball PRIVATE dxball_core winmm user32 gdi32)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(dxball PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_compile_options(dxball_state_reader PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_compile_options(dxball_ddraw_loss_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_compile_options(dxball_resource_reader PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
